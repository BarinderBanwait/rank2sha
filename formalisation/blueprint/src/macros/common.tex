% The theorem-like environments defined below are those that appear by default
% in the dependency graph. See the README of leanblueprint if you need help to
% customize this. 
% The configuration below use the theorem counter for all those environments
% (this is what the [theorem] arguments mean) and never resets it.
% If you want for instance to number them within chapters then you can add
% [chapter] at the end of the next line.
\newtheorem{theorem}{Theorem}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}

\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}

% --- The classical-input / assumption environment ------------------------
% Interface fields of `ClassicalInputs` (citable classical theorems: Rubin,
% Mazur, Greenberg, MTT, Bannai–Kobayashi, de Shalit, Schneider/Perrin-Riou,
% Hasse) are rendered in this environment rather than as
% `lemma`/`proposition`/`theorem`, so a reader can see at a glance, by the
% header word alone ("Assumption" vs "Lemma"/"Theorem"), which statements are
% asserted on citation and which are Lean-kernel-proved. `\leanok` is never
% applied to a node in this environment: these are the human-refereeable
% trust boundary, not proof obligations.
%
% plasTeX's `plastexdepgraph` plugin draws dependency-graph nodes only for
% theorem *kinds* listed in the `thms=` option passed to
% `\usepackage[showmore, dep_graph]{blueprint}` in `src/web.tex`, whose
% default is `definition+lemma+proposition+theorem+corollary`. `src/web.tex`
% lists `assumption` there (added 2026-08-18), so assumption nodes do appear
% in the graph. An earlier note here said they did not, and described a
% workaround of declaring them as `definition` instead; that note was stale
% by the time it was read, and was removed on 2026-08-29 (R3bp). Nodes in
% this environment never carry `\leanok`, so they never receive the green
% fill a proved node gets.
\newtheorem{assumption}[theorem]{Assumption}

% --- R3bp (2026-08-29): the paper's own conjecture/hypothesis environments.
% conj:strong, conj:weak and conj:EK are Conjectures in the paper and
% hyp:sinnott is a Hypothesis; rendering them as `definition` or `assumption`
% made every cross-reference read "Definition N" or "Assumption N" where the
% surrounding prose says "Conjecture" / "Hypothesis". Both share the theorem
% counter, as every other environment here does, and both are listed in the
% `thms=` option of src/web.tex so they appear as dependency-graph nodes.
%
% \leanok discipline is unchanged and reads the same way in these two
% environments as everywhere else: it marks that the Lean *statement* matches,
% not that anything is proved. conj:strong and conj:weak carry it (ConjStrong
% and ConjWeak are Prop-valued defs whose statements match); conj:EK and
% hyp:sinnott do not (see their own text for the reason in each case).
\newtheorem{conjecture}[theorem]{Conjecture}
\newtheorem{hypothesis}[theorem]{Hypothesis}
